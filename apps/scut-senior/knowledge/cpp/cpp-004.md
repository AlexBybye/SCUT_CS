	template <typename T>
	class Node {
	public:
		T data;
		Node<T>* next;

		Node(T value) : data(value), next(nullptr) {}
	};

	template <typename T>
	class LinkedList {
	private:
		Node<T>* head;
		Node<T>* tail;
	public:
		LinkedList() : head(nullptr), tail(nullptr) {}

		void insert(T value) {
			Node<T>* newNode = new Node<T>(value);
			if (head == nullptr) {
				head = tail = newNode;
			}
			else {
				tail->next = newNode;
				tail = newNode;
			}
		}

		void display() {
			Node<T>* current = head;
			while (current != nullptr) {
				std::cout << current->data << std::endl;
				current = current->next;
			}
		}
	};
